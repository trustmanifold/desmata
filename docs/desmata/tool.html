 23def short(line: str) -> str:
 24    """
 25    Don't overwhelm the user with text, truncate lines with '...'
 26    """
 27    if len(line) < max_chars:
 28        return line
 29    else:
 30        return line[: max_chars - 100] + "..." + line[-100:]

 33def pretty(it: str | None, prefix: str = "") -> str:
 34    """
 35    Don't overwhelm the user with text, omit lines with '...'
 36    """
 37    if it is None or it.strip() == "":
 38        return ""
 39    else:
 40        lines = [x for x in it.splitlines() if x.strip() != ""]
 41        if len(lines) < max_lines:
 42            return indent("\n".join(map(short, lines)), prefix=prefix)
 43        return indent(
 44            "\n".join(map(short, [*lines[: max_lines - 5], "...", *lines[-5:]])),
 45            prefix=prefix,
 46        )

 49def env_pretty(inner: dict[str, str], outer: dict[str, str]) -> dict[str, str]:
 50    """
 51    Only log env vars if they differ from the enclosing environment.
 52    """
 53    show = {}
 54    for k, v in inner.items():
 55        if k == "KUBECONFIG":
 56            try:
 57                outer_v = outer[k]
 58            except KeyError:
 59                show[k] = v
 60            else:
 61                if v != outer_v:
 62                    show[k] = v
 63    return show

 66def log_pretty(it: str) -> str:
 67    return short(re.sub(r"\s*\n\s*", ";", it.strip() or ""))

 70def too_long(it: str | None) -> bool:
 71    lines = (it or "").splitlines()
 72    if len(lines) >= max_lines:
 73        return True
 74    if any(len(x) > max_chars for x in lines):
 75        return True
 76    return False

145class Tool:
146    """
147    A helper for running commands. You can bind a fixed context when you initialize this
148    class (i.e. the things that do not change between calls).  The interesting parts
149    can be supplied on a per-call basis.
150
151        # fixed context
152        kubectl = Tool(name="kubectl",
153                       env_filter = lambda e: dict(e, KUBECONFIG="/some/path"),
154                       cmd_filter = lambda c: ["-n", "mynamespace", *c])
155
156        # dynamic context
157        kubectl(["get", "pods"])
158    """
159
160    log: Logger
161    name: str
162    path: Path | None = None
163    cwd: Path | str | None = None
164    cmd_filter: Callable[[Cmd], Cmd] | None = None
165    env_filter: Callable[[EnvVars], EnvVars] | None = None
166    show_stdin: bool = True
167    show_stdout: bool = True
168
169    def _print(self, string: str, prefix: str = "") -> None:
170        if self.show_stdin or self.show_stdout:
171            print(pretty(string, prefix=prefix), file=sys.stderr)
172
173    def _print_stdin(self, string: str | None) -> None:
174        if self.show_stdout and string and string.strip():
175            self._print(string, prefix=f"  {self.name}<<  ")
176
177    def _print_stdout(self, string: str) -> None:
178        if self.show_stdout and string and string.strip():
179            self._print(string, prefix=f"  {self.name}>>  ")
180
181    def _build_cmd(self, *args: str) -> list[str]:
182        if self.path:
183            self.path.resolve()
184            assert self.path.exists()
185            name = str(self.path.absolute())
186        else:
187            name = self.name
188
189        suffix = self.cmd_filter(list(args)) if self.cmd_filter else list(args)
190        if isinstance(suffix, str):
191            suffix = [suffix]
192        return [name, *suffix]
193
194    def get_popen_kwargs(self, **kwargs: Any) -> dict[str, Any]:
195        """
196        subprocess.Popen args can be bound to the tool object, or provided at
197        call time via **kwargs.  Call this to move them to kwargs from the
198        object and provide any other handling specific to that kwarg.
199        """
200
201        try:
202            cwd = kwargs["cwd"]
203        except KeyError:
204            cwd = self.cwd
205            if isinstance(self.cwd, Path):
206                kwargs["cwd"] = str(self.cwd.resolve())
207            else:
208                kwargs["cwd"] = self.cwd
209
210        if cwd and isinstance(cwd, Path):
211            kwargs["cwd"] = str(cwd.resolve())
212
213        return kwargs

215    def run(
216        self,
217        *args: str,
218        env_extras: EnvVars = {},
219        stdin: str | None = None,
220        tolerate_err: bool = True,
221        no_log: bool = False,
222        no_stdout: bool = False,
223        no_stdin: bool = False,
224        **kwargs: Any,
225    ) -> CompletedCommand:
226        logfunc = self.log.getChild(self.name).debug
227        if no_log:
228            logfunc = Mock()
229
230        # build cmd
231        cmd = self._build_cmd(*args)
232
233        # finalize Popen kwargs
234        popen_kwargs = self.get_popen_kwargs(**kwargs)
235
236        # build env
237        outer_env = dict(os.environ)
238        env = self.env_filter(outer_env.copy()) if self.env_filter else outer_env
239        env.update(env_extras)
240
241        # communicate attempt
242        pending = PendingCommand(
243            cmd=cmd, env=env, stdin=stdin or "", outer_env=outer_env, kwargs=popen_kwargs
244        )
245        logfunc(pending)
246        pending.check()
247        if not no_stdin:
248            self._print_stdin(stdin)
249
250        # pass stderr though, handle stdin and stdout explicitly
251        process = subprocess.Popen(
252            cmd,
253            env=env,
254            stdin=subprocess.PIPE,
255            stdout=subprocess.PIPE,
256            stderr=sys.stderr,
257            text=True,
258            **popen_kwargs,
259        )
260
261        # run it, provide stdin, get stdout
262        stdout, _ = process.communicate(input=stdin)
263
264        # communicate results
265        completed = CompletedCommand(stdout, exit_code=process.returncode)
266        logfunc(completed)
267        if not no_stdout:
268            if completed.long_stdout():
269                self._print_stdout(stdout)
270            self._print("\n")
271
272        if not tolerate_err and completed.exit_code != 0:
273            raise subprocess.CalledProcessError(completed.exit_code, cmd, output=stdout)
274
275        return completed

304    def run_to_file(
305        self,
306        *args: str,
307        dest: Path,
308        env_extras: EnvVars = {},
309        tolerate_err: bool = False,
310        **kwargs: Any,
311    ) -> None:
312        """
313        Run the tool and stream its (binary) stdout straight into ``dest``.
314
315        Unlike :meth:`run`, this never decodes stdout as text, so it is safe for
316        binary payloads like NAR streams (``nix-store --export``) and CAR files
317        (``ipfs dag export``). stderr is passed through as usual.
318        """
319        cmd = self._build_cmd(*args)
320        env = self._binary_env(env_extras)
321        popen_kwargs = self.get_popen_kwargs(**kwargs)
322        self.log.getChild(self.name).debug(
323            json.dumps({"pending(->file)": {"cmd": cmd, "dest": str(dest)}})
324        )
325        with open(dest, "wb") as out:
326            process = subprocess.Popen(
327                cmd, env=env, stdout=out, stderr=sys.stderr, **popen_kwargs
328            )
329            process.communicate()
330        if not tolerate_err and process.returncode != 0:
331            raise subprocess.CalledProcessError(process.returncode, cmd)
332
333    def run_from_file(
334        self,
335        *args: str,
336        src: Path,
337        env_extras: EnvVars = {},
338        tolerate_err: bool = False,
339        **kwargs: Any,
340    ) -> str:
341        """
342        Run the tool with ``src`` streamed (as binary) into its stdin, returning
343        decoded stdout.
344
345        The inverse of :meth:`run_to_file`: used for ``nix-store --import``,
346        whose input is a binary NAR but whose output (the imported store paths)
347        is text.
348        """
349        cmd = self._build_cmd(*args)
350        env = self._binary_env(env_extras)
351        popen_kwargs = self.get_popen_kwargs(**kwargs)
352        self.log.getChild(self.name).debug(
353            json.dumps({"pending(<-file)": {"cmd": cmd, "src": str(src)}})
354        )
355        with open(src, "rb") as inp:
356            process = subprocess.Popen(
357                cmd,
358                env=env,
359                stdin=inp,
360                stdout=subprocess.PIPE,
361                stderr=sys.stderr,
362                **popen_kwargs,
363            )
364            stdout_bytes, _ = process.communicate()
365        if not tolerate_err and process.returncode != 0:
366            raise subprocess.CalledProcessError(process.returncode, cmd)
367        return stdout_bytes.decode()
