 33    class IPFS(Tool):
 34        """The ipfs/kubo CLI as a desmata tool.
 35
 36        Implements the ``ContentBackend`` protocol (see :mod:`desmata.content`):
 37        the methods under "ContentBackend" below speak backend-tagged
 38        :class:`Hash` values, while the rest of the surface (``add``, ``dag_*``,
 39        ``get``, ...) is the backend-native layer that traffics in raw CIDs.
 40        """
 41
 42        backend = Backend.ipfs
 43
 44        def __init__(self, root: Path, context: CellContext):
 45            loggers = context.loggers.specialize("ipfs")
 46            ipfs_path_entry = root / "bin"
 47            ipfs_exe = ipfs_path_entry / "ipfs"
 48            super().__init__(
 49                name="ipfs",
 50                path=ipfs_exe,
 51                log=loggers.proc,
 52                env_filter=context.get_env_filter(exec_path=ipfs_path_entry),
 53            )
 54
 55        def get_hash(self, target: Path) -> str:
 56            output = self("add", "-r", "--only-hash", str(target.resolve()))
 57            # sample output:
 58            #   added QmWfbz6Tvds3X2y3iUv994ootBQ8JdyspiEqYXtAVHPfVB builtins/flake.lock
 59            #   added QmcA67vzYhWSCBB3KKFFtTbyVxy349SRQpzUF4Be8r4hft builtins/flake.nix
 60            #   added QmS2CjUTboH59Pfz2BwRFJpBbQboAiqQvyoq3wPF9e9Wwf builtins
 61            # Take the last hash, which will be the toplevel dir
 62            # (or just the file if the target wasn't a dir)
 63            return output.splitlines()[-1].split()[1]
 64
 65        # --- ContentBackend --------------------------------------------------
 66        # the backend-agnostic surface: desmata's protocols (hashing a
 67        # dependency, publishing/fetching content) speak Hash, not raw CIDs.
 68
 69        def hash_path(self, path: Path) -> Hash:
 70            """Content-address ``path`` without storing it (offline, deterministic)."""
 71            return Hash(backend=self.backend, digest=self.get_hash(path))
 72
 73        def publish(self, path: Path) -> Hash:
 74            """Store ``path`` in this repo and return its content address."""
 75            cid = self.add(path, recursive=path.is_dir())
 76            return Hash(backend=self.backend, digest=cid)
 77
 78        def fetch(self, hash: Hash, into: Path) -> None:
 79            """Materialize the content addressed by ``hash`` at ``into``."""
 80            if not self.can_handle(hash):
 81                raise UnknownBackendException(f"ipfs cannot resolve {hash}")
 82            self.get(hash.digest, into)
 83
 84        def can_handle(self, hash: Hash) -> bool:
 85            return hash.backend is Backend.ipfs
 86
 87        # --- content-addressed store/move -----------------------------------
 88        # get_hash above only *computes* a CID (``--only-hash``); the methods
 89        # below actually store and move bytes between repos. Each Tools.IPFS is
 90        # bound (via its context's HOME) to its own ipfs repo, so two instances
 91        # on different homes are two independent peers. All operations are
 92        # ``--offline`` -- the partition-tolerant data path never touches the
 93        # network; moving blocks between repos is done with a CAR file.
 94
 95        def init(self) -> None:
 96            """Initialize this repo if it isn't already (idempotent)."""
 97            self("init", tolerate_err=True)
 98
 99        def add(
100            self, target: Path, *, recursive: bool = False, offline: bool = True
101        ) -> str:
102            """Store ``target`` in this repo and return its (root) CID.
103
104            ``recursive`` is needed when ``target`` is a directory (e.g. a nix
105            store path); ``-Q`` returns only the top-level CID."""
106            args = ["add", "-Q"]
107            if recursive:
108                args += ["-r", "-H"]  # include directory contents + hidden files
109            if offline:
110                args.append("--offline")
111            return self(*args, str(target.resolve())).strip().splitlines()[-1]
112
113        def refs(self, cid: str, *, offline: bool = True) -> list[str]:
114            """The CIDs of every distinct block reachable under ``cid`` (its
115            descendants, not ``cid`` itself). The block count of a stored object
116            is ``len(refs(cid)) + 1``."""
117            args = ["refs", "-r", "-u"]
118            if offline:
119                args.append("--offline")
120            out = self(*args, cid)
121            return [line.strip() for line in out.splitlines() if line.strip()]
122
123        def ls(self, cid: str) -> list[tuple[str, int, str]]:
124            """One level of the UnixFS DAG under ``cid`` as (child_cid, size,
125            name) tuples. Directory entries have names (``bin/``); a chunked
126            file's internal nodes come back with an empty name."""
127            entries: list[tuple[str, int, str]] = []
128            for line in self("ls", cid).splitlines():
129                if not line.strip():
130                    continue
131                parts = line.split(maxsplit=2)
132                child = parts[0]
133                try:
134                    size = int(parts[1]) if len(parts) > 1 else 0
135                except ValueError:
136                    size = 0  # directories report '-'
137                name = parts[2] if len(parts) > 2 else ""
138                entries.append((child, size, name))
139            return entries
140
141        def dag_export(self, cid: str, dest: Path) -> None:
142            """Write the blocks under ``cid`` to a CAR file at ``dest``."""
143            self.run_to_file("dag", "export", cid, dest=dest)
144
145        def dag_import(self, car: Path) -> None:
146            """Load a CAR file's blocks into this repo."""
147            self("dag", "import", str(car.resolve()))
148
149        def dag_put(self, obj: dict) -> str:
150            """Store ``obj`` as an IPLD node (dag-cbor) and return its CID. Links
151            are expressed as ``{"/": "<cid>"}``; ``dag_export`` of the result
152            follows them, so a manifest that links per-path NARs exports them
153            all in one CAR."""
154            return self(
155                "dag", "put",
156                "--input-codec", "dag-json", "--store-codec", "dag-cbor",
157                stdin=json.dumps(obj),
158            ).strip()
159
160        def dag_get(self, cid: str) -> dict:
161            """Read an IPLD node back as dag-json (links as ``{"/": cid}``)."""
162            return json.loads(self("dag", "get", cid))
163
164        def get(self, cid: str, dest: Path, *, offline: bool = True) -> None:
165            """Materialize ``cid`` from this repo's blocks into ``dest``."""
166            args = ["get"]
167            if offline:
168                args.append("--offline")
169            self(*args, cid, "-o", str(dest))

173    class IPFS(Dependency):
174        @staticmethod
175        def build_or_get(context: CellContext) -> "Deps.IPFS":
176            nix = get_nix(context)
177
178            # ensure ipfs exists externally
179            _root, _deps = nix.build("ipfs")
180            ipfs_tool = Tools.IPFS(root=_root, context=context)
181
182            # `ipfs init` creates a repo (keys + config) under the cell's HOME
183            # and errors if run a second time. Only initialize when absent so
184            # build_or_get stays idempotent: re-bootstrapping, and building any
185            # cell that depends on builtins, must not fail just because ipfs is
186            # already initialized. Use `dsm clean ipfs keys` to force a fresh repo.
187            ipfs_repo = Path(context.home) / ".ipfs"
188            if not (ipfs_repo / "config").exists():
189                ipfs_tool("init")
190
191            # bring it and its deps under desmata's control
192            deps_by_id: dict[str, "Deps.IPFS"] = {}
193            root_dep: "Deps.IPFS | None" = None
194            for subdag in nix.dep_dags(_deps, str(_root)):
195                proto_dep = ProtoDependency(
196                    target=subdag.info.path,
197                    dependencies=[
198                        (nix.get_id(x.info.path), x.info.path)
199                        for x in subdag.immediate_dependencies
200                    ],
201                )
202                dep_id, dep_hash = context.internalize_ids_hashes(
203                    proto_dep=proto_dep,
204                    id_getter=nix.get_id,
205                    hasher=ipfs_tool,
206                )
207
208                # nix.dep_dags provides leaves first and works towards the root,
209                # so each immediate dependency already has a Dependency in
210                # deps_by_id; resolve the /nix/store deps to Dependencies.
211                immediate_dependencies: dict[str, Dependency] = {}
212                for child in subdag.immediate_dependencies:
213                    child_id = nix.get_id(child.info.path)
214                    immediate_dependencies[child_id] = deps_by_id[child_id]
215
216                dependency = Deps.IPFS(
217                    id=dep_id,
218                    hash=dep_hash,
219                    root=subdag.info.path,
220                    immediate_dependencies=immediate_dependencies,
221                )
222                deps_by_id[dep_id] = dependency
223                # dep_dags yields the root last
224                root_dep = dependency
225
226            if root_dep is None:
227                raise ValueError("Nothing to build for ipfs")
228
229            return root_dep
230
231        def get_tool(self, context: CellContext) -> "Tools.IPFS":
232            return Tools.IPFS(root=Path(self.root), context=context)

235class BuiltinsClosure(Closure):
236    # ipfs is the only *managed* builtin dependency: desmata builds it, content
237    # addresses it, and (eventually) shares it peer-to-peer. nix and git are
238    # trusted bootstrap tools the user is expected to have installed; their
239    # conformance is checked via their interfaces, not managed here.
240    ipfs: Deps.IPFS

# fixed context
kubectl = Tool(name="kubectl",
               env_filter = lambda e: dict(e, KUBECONFIG="/some/path"),
               cmd_filter = lambda c: ["-n", "mynamespace", *c])

# dynamic context
kubectl(["get", "pods"])
