113        but only if they're on the list.

116        for var, val in os.environ.items():

130    def internalize_ids_hashes(
131        self,
132        *,
133        proto_dep: ProtoDependency,
134        id_getter: IdGetter,
135        hasher: ContentBackend,
136    ) -> tuple[DependencyId, DependencyHash]:
137        """
138        Internalizes a dependency from an external path to the desmata-controlled storage.
139        Creates both id-based and hash-based references for the dependency.
140        Returns the ID and the content address.
141        """
142
143        # Get dependency ID and prepare paths
144        dep_id = id_getter(proto_dep.target)
145        id_path = self._userspace.deps_by_id / dep_id
146
147        # If the id_path already exists, we just need to ensure the hash path exists
148        if id_path.exists():
149            hash_val = hasher.hash_path(id_path)
150            hash_path = self._userspace.deps_by_hash / hash_val.dirname
151
152            # Create the hash path if it doesn't exist
153            if not hash_path.exists():
154                create_hard_links(id_path, hash_path)
155
156            return DependencyId(dep_id), DependencyHash(hash_val)
157        
158        # Create the id directory
159        id_path.mkdir(parents=True, exist_ok=True)
160        
161        # Process the target path
162        target_path = Path(proto_dep.target)
163        
164        # Check if target is read-only and on the same device (likely in nix store)
165        target_stat = os.stat(target_path)
166        data_stat = os.stat(self._userspace.data)
167        is_readonly = not bool(target_stat.st_mode & stat.S_IWUSR)
168        same_device = target_stat.st_dev == data_stat.st_dev
169        
170        # Either create hard links (if safe) or copy the files
171        if is_readonly and same_device:
172            create_hard_links(target_path, id_path)
173        else:
174            # Not in nix store or on different device - copy the data
175            if target_path.is_dir():
176                shutil.copytree(target_path, id_path, dirs_exist_ok=True)
177            else:
178                shutil.copy2(target_path, id_path / target_path.name)
179        
180        # Create the hash-based path
181        hash_val = hasher.hash_path(id_path)
182        hash_path = self._userspace.deps_by_hash / hash_val.dirname
183        
184        # Create hard links from id_path to hash_path
185        if not hash_path.exists():
186            create_hard_links(id_path, hash_path)
187
188        return DependencyId(dep_id), DependencyHash(hash_val)
189
190    def get_env_filter(
191        self,
192        *,
193        exec_path: Path | list[Path] = [],
194        passthru_vars: list[str] = default_passthrough_vars,
195        set_default_env: bool = True,
196        env_overrides: EnvVars = {},
197    ) -> EnvFilter:
198        passthru = self._get_passed_thru_vars(passthru_vars)
199        env_vars = self._get_default_env() if set_default_env else {}
200
201        match exec_path:
202            case Path():
203                deps = str(exec_path)
204            case list():
205                deps = list(map(str, exec_path))
206
207        def filter(env: EnvVars) -> EnvVars:
208            inner_env = env_vars.copy()
209            inner_env.update(self._get_passed_thru_vars(passthru))
210            if exec_path := inner_env.get("PATH"):
211                exec_path = ":".join([*exec_path.split(":"), *deps])
212            return inner_env
213
214        return filter

256    @staticmethod
257    def _get_closure_type(CellType: type[SpecificCell]) -> type[Closure]:
258        closure_types = []
259        for base in CellType.__orig_bases__:
260            if hasattr(base, "__args__"):
261                for arg in base.__args__:
262                    if issubclass(arg, Closure):
263                        closure_types.append(arg)
264
265        if len(closure_types) != 1:
266            raise BadCellClassException(
267                f"Expected exactly one closure type for "
268                f"cell type {CellType}, instead got {closure_types}. "
269                "Cells should have Cell[SomeClosure] as a base class, "
270                "where SomeClosure has Closure as a base class"
271            )
272        return closure_types[0]
273
274    @staticmethod
275    def _get_dependency_types(ClosureType: type[Closure]) -> dict[str, Dependency]:
276        dependency_types: dict[str, Dependency] = {}
277        for attr, annotated_type in ClosureType.__annotations__.items():
278            if issubclass(annotated_type, Dependency):
279                dependency_types[attr] = annotated_type
280        return dependency_types

282    def get(self, CellType: type[SpecificCell]) -> SpecificCell:
283        # build a cell context
284        module_name = CellType.__module__
285        module = importlib.import_module(module_name)
286        cell_file = Path(module.__file__)
287        name = self._cell_name(cell_file)
288        inode, device_id = self._cell_file_inode_device(cell_file)
289        context = BasicContext(
290            name=name,
291            cell_dir=cell_file.parent,
292            userspace=self.userspace,
293            loggers=self.log,
294        )
295        self.log.msg.debug(f"CONTEXT, {context.__dict__}")
296
297        # get a content backend to hash with (ipfs is the only one so far)
298        hasher: ContentBackend
299        ipfs_dep: DesmataBuiltinDeps.IPFS
300        if CellType is DesmataBuiltins:
301            ipfs_dep = DesmataBuiltinDeps.IPFS.build_or_get(context)
302            hasher = DesmataBuiltinTools.IPFS(root=ipfs_dep.root, context=context)
303        else:
304            builtins = self.get(DesmataBuiltins)
305            hasher = builtins.ipfs
306            ipfs_dep = builtins.closure.ipfs
307        self.backends.register(hasher)
308
309        # populate the closure
310        deps: dict[str, Dependency] = {}
311        ClosureType = DefaultCellFactory._get_closure_type(CellType)
312        for attr_name, AttrType in DefaultCellFactory._get_dependency_types(
313            ClosureType
314        ).items():
315            dep: Dependency
316
317            # populate dep.root
318            if AttrType is DesmataBuiltinDeps.IPFS and attr_name == "ipfs":
319                dep = ipfs_dep
320            else:
321                dep = AttrType.build_or_get(context, hasher=hasher)
322
323            # populate dep.root and dep.id based on dep.root
324            dep.hash = DependencyHash(hasher.hash_path(dep.root))
325            dep.id = dep.get_id(dep.root)
326            deps[attr_name] = dep
327
328        closure = ClosureType(local_name=name, **deps)
329
330        # capture build provenance: a Trustix-projectable narinfo per store path
331        # in each managed dependency's closure (see provenance.py / phase-2.md).
332        nix = get_nix(context)
333        records: list = []
334        for dep in deps.values():
335            records.extend(provenance.closure_provenance(nix, dep))
336        provenance.save(self.userspace, records)
337
338        return CellType(closure, context)

Base class for protocol classes.

Protocol classes are defined as::

class Proto(Protocol):
    def meth(self) -> int:
        ...

Such classes are primarily used with static type checkers that recognize structural subtyping (static duck-typing).

For example::

class C:
    def meth(self) -> int:
        return 0

def func(x: Proto) -> int:
    return x.meth()

func(C())  # Passes static type check

See PEP 544 for details. Protocol classes decorated with @typing.runtime_checkable act as simple-minded runtime protocols that check only the presence of given attributes, ignoring their type signatures. Protocol classes can be generic, they are defined as::

class GenProto[T](Protocol):
    def meth(self) -> T:
        ...

171        # Either create hard links (if safe) or copy the files
172        if is_readonly and same_device:
173            create_hard_links(target_path, id_path)
174        else:
175            # Not in nix store or on different device - copy the data
176            if target_path.is_dir():
177                shutil.copytree(target_path, id_path, dirs_exist_ok=True)
178            else:
179                shutil.copy2(target_path, id_path / target_path.name)
180        
181        # Create the hash-based path
182        hash_val = hasher.hash_path(id_path)
183        hash_path = self._userspace.deps_by_hash / hash_val.dirname
184        
185        # Create hard links from id_path to hash_path
186        if not hash_path.exists():
187            create_hard_links(id_path, hash_path)
188
189        return DependencyId(dep_id), DependencyHash(hash_val)

Base class for protocol classes.

Protocol classes are defined as::

class Proto(Protocol):
    def meth(self) -> int:
        ...

Such classes are primarily used with static type checkers that recognize structural subtyping (static duck-typing).

For example::

class C:
    def meth(self) -> int:
        return 0

def func(x: Proto) -> int:
    return x.meth()

func(C())  # Passes static type check

See PEP 544 for details. Protocol classes decorated with @typing.runtime_checkable act as simple-minded runtime protocols that check only the presence of given attributes, ignoring their type signatures. Protocol classes can be generic, they are defined as::

class GenProto[T](Protocol):
    def meth(self) -> T:
        ...
