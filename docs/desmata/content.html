desmata.content

Self-describing content addresses and the backends that serve them.

Desmata addresses cells and dependencies by hash. Today every hash is an IPFS CID, but the hash format should not assume that forever (see agent_primers/iroh.md): a hash must carry enough information that anyone who encounters it can tell which backend produced it and therefore how to resolve it. This module is the one place that knows all the schemes:

  • Backend -- the enum of content-addressing backends desmata knows.
  • Hash -- a backend-tagged content address with a canonical string form, dsm:<backend>:<digest>.
  • ContentBackend -- the protocol a backend implementation satisfies (hash a path offline, publish it, fetch by hash).
  • BackendRegistry -- dispatch from a Hash to the ContentBackend that can serve it.

This module must stay independent of the rest of desmata (stdlib + pydantic only) so that anything -- protocols, tools, tests -- can import it freely.

 42class Hash(BaseModel):
 43    """A self-describing content address.
 44
 45    :param backend: which backend produced (and can resolve) this hash.
 46    :param digest: the backend-native digest string -- for IPFS, a CID.
 47
 48    The canonical string form is ``dsm:<backend>:<digest>`` (e.g.
 49    ``dsm:ipfs:QmW3J3...``); :meth:`parse` is its inverse. Instances are frozen
 50    (hashable, usable as dict keys) and compare by value, so "same backend, same
 51    digest" means equal.
 52    """
 53
 54    model_config = ConfigDict(frozen=True)
 55
 56    backend: Backend
 57    digest: str
 58
 59    def __str__(self) -> str:
 60        return f"{SCHEME}:{self.backend}:{self.digest}"
 61
 62    @property
 63    def dirname(self) -> str:
 64        """A filesystem-safe form (no ``:``), used for content-addressed
 65        directory names such as ``deps/by_hash/<dirname>``."""
 66        return f"{self.backend}-{self.digest}"
 67
 68    @classmethod
 69    def parse(cls, text: str) -> "Hash":
 70        """The inverse of ``str(hash)``.
 71
 72        :raises ValueError: if ``text`` is not ``dsm:<backend>:<digest>``.
 73        :raises UnknownBackendException: if the backend tag isn't one desmata knows.
 74        """
 75        scheme, _, rest = text.partition(":")
 76        backend_name, _, digest = rest.partition(":")
 77        if scheme != SCHEME or not backend_name or not digest:
 78            raise ValueError(
 79                f"expected a hash of the form '{SCHEME}:<backend>:<digest>', got {text!r}"
 80            )
 81        try:
 82            backend = Backend(backend_name)
 83        except ValueError:
 84            raise UnknownBackendException(
 85                f"unknown content backend {backend_name!r} in {text!r} "
 86                f"(known: {', '.join(Backend)})"
 87            )
 88        return cls(backend=backend, digest=digest)

 91@runtime_checkable
 92class ContentBackend(Protocol):
 93    """What a content-addressing backend must be able to do.
 94
 95    The generalization of the old ``PathHasher``: hashing and moving content are
 96    two halves of the same capability, so they live behind one protocol. IPFS
 97    (``desmata.builtins.cell.Tools.IPFS``) is the sole implementation today.
 98    """
 99
100    backend: Backend
101
102    def hash_path(self, path: Path) -> Hash:
103        """Content-address ``path`` without storing it. Must be deterministic
104        and work offline -- this is what makes 'the same bytes hash to this
105        everywhere' true."""
106        raise NotImplementedError()
107
108    def publish(self, path: Path) -> Hash:
109        """Store ``path``'s content in this backend and return its address."""
110        raise NotImplementedError()
111
112    def fetch(self, hash: Hash, into: Path) -> None:
113        """Materialize the content addressed by ``hash`` at ``into``."""
114        raise NotImplementedError()
115
116    def can_handle(self, hash: Hash) -> bool:
117        """Whether this backend can resolve ``hash`` (i.e. produced its scheme)."""
118        raise NotImplementedError()

121class BackendRegistry:
122    """Dispatch from a :class:`Hash` to the backend that can serve it.
123
124    One entry today, but resolving 'which backend handles this hash?' through
125    the registry (rather than assuming IPFS) is what keeps the door open for a
126    second backend.
127    """
128
129    def __init__(self) -> None:
130        self._backends: dict[Backend, ContentBackend] = {}
131
132    def register(self, impl: ContentBackend) -> None:
133        self._backends[impl.backend] = impl
134
135    def get(self, backend: Backend) -> ContentBackend:
136        try:
137            return self._backends[backend]
138        except KeyError:
139            raise UnknownBackendException(
140                f"no {backend} backend registered "
141                f"(registered: {', '.join(self._backends) or 'none'})"
142            )
143
144    def for_hash(self, hash: Hash) -> ContentBackend:
145        return self.get(hash.backend)

1768    cls = type(self)
1769
1770    if cls._is_protocol:
1771        raise TypeError('Protocols cannot be instantiated')
1772
1773    # Already using a custom `__init__`. No need to calculate correct
1774    # `__init__` to call. This can lead to RecursionError. See bpo-45121.
1775    if cls.__init__ is not _no_init_or_replace_init:
1776        return
1777
1778    # Initially, `__init__` of a protocol subclass is set to `_no_init_or_replace_init`.
1779    # The first instantiation of the subclass will call `_no_init_or_replace_init` which
1780    # searches for a proper new `__init__` in the MRO. The new `__init__`
1781    # replaces the subclass' old `__init__` (ie `_no_init_or_replace_init`). Subsequent
1782    # instantiation of the protocol subclass will thus use the new
1783    # `__init__` and no longer call `_no_init_or_replace_init`.
1784    for base in cls.__mro__:
1785        init = base.__dict__.get('__init__', _no_init_or_replace_init)
1786        if init is not _no_init_or_replace_init:
1787            cls.__init__ = init
1788            break
1789    else:
1790        # should not happen
1791        cls.__init__ = object.__init__
1792
